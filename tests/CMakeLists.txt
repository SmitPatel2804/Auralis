function(auralis_add_test name)
    add_executable("${name}" ${ARGN})
    target_link_libraries("${name}" PRIVATE Qt6::Test)
    auralis_apply_warnings("${name}")
    add_test(NAME "${name}" COMMAND "${name}")
endfunction()
